cmake_minimum_required(VERSION 3.22)
include(cmake/Identity.cmake)
project(MasterMonitor VERSION ${MM_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE 8.0.9 checkout")
set(EBUR128_SOURCE_DIR "" CACHE PATH "Existing libebur128 1.2.6 checkout")
if(NOT JUCE_SOURCE_DIR)
 FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG ${MM_JUCE_REV})
 FetchContent_MakeAvailable(juce)
else()
 add_subdirectory(${JUCE_SOURCE_DIR} juce)
endif()
if(NOT EBUR128_SOURCE_DIR)
 FetchContent_Declare(ebur_source GIT_REPOSITORY https://github.com/jiixyj/libebur128.git GIT_TAG ${MM_EBUR_REV})
 FetchContent_GetProperties(ebur_source)
 if(NOT ebur_source_POPULATED)
  FetchContent_Populate(ebur_source)
 endif()
 set(EBUR128_SOURCE_DIR ${ebur_source_SOURCE_DIR})
endif()
# Compile the pinned small library directly; avoid its legacy CMake policies.
add_library(ebur STATIC ${EBUR128_SOURCE_DIR}/ebur128/ebur128.c)
target_include_directories(ebur PUBLIC ${EBUR128_SOURCE_DIR}/ebur128 PRIVATE ${EBUR128_SOURCE_DIR}/ebur128/queue)
if(MSVC)
 target_compile_definitions(ebur PRIVATE _USE_MATH_DEFINES)
endif()
set(formats VST3 Standalone)
if(APPLE)
 list(APPEND formats AU)
endif()
juce_add_binary_data(MMAssets SOURCES assets/logo.png)
juce_add_plugin(MasterMonitor COMPANY_NAME "${MM_MANUFACTURER}" BUNDLE_ID "${MM_BUNDLE_ID}"
 PLUGIN_MANUFACTURER_CODE ${MM_MANUFACTURER_CODE} PLUGIN_CODE ${MM_PLUGIN_CODE}
 PRODUCT_NAME "MASTER MONITOR" FORMATS ${formats} IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE
 NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
 VST3_CATEGORIES Fx Analyzer AU_MAIN_TYPE kAudioUnitType_Effect
 MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Optional audio input for metering; input is muted on first launch.")
set(MM_SOURCES src/Analysis.cpp src/Processor.cpp src/Editor.cpp)
target_sources(MasterMonitor PRIVATE ${MM_SOURCES})
target_include_directories(MasterMonitor PUBLIC src)
target_compile_definitions(MasterMonitor PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
target_link_libraries(MasterMonitor PUBLIC ebur MMAssets juce::juce_audio_utils juce::juce_dsp PRIVATE juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(MMTests PRODUCT_NAME MMTests)
target_sources(MMTests PRIVATE tests/Tests.cpp)
target_link_libraries(MMTests PRIVATE MasterMonitor)
add_test(NAME deterministic COMMAND MMTests)
set_tests_properties(deterministic PROPERTIES ENVIRONMENT "MM_NO_USER_SETTINGS=1")
juce_add_console_app(MMRender PRODUCT_NAME MMRender)
target_sources(MMRender PRIVATE tests/Render.cpp)
target_link_libraries(MMRender PRIVATE MasterMonitor)

# Test utilities place the large preallocated analyser on the main stack.
if(MSVC)
 target_link_options(MMTests PRIVATE /STACK:16777216)
 target_link_options(MMRender PRIVATE /STACK:16777216)
endif()
